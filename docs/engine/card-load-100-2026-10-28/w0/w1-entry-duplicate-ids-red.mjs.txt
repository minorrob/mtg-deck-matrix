import assert from "node:assert/strict";
import {readFileSync} from "node:fs";
import {createState,addObject} from "../w0-w1-faces/game/engine/state/index.mjs";
import {compileScript} from "../w0-w1-faces/game/engine/cards/index.mjs";
import {beginRuleOperation,answerRuleOperation} from "../w0-w1-faces/game/engine/rules/operations.mjs";
import {afterResolving} from "../w0-w1-faces/game/engine/rules/priority.mjs";
import {awaitingChoice,resolveAwaiting} from "../w0-w1-faces/game/engine/rules/turn.mjs";
let state=createState({players:[{name:"A"},{name:"B"},{name:"C"}]});Object.assign(state,{phase:"MAIN1",turn:1,activePlayer:0,priorityPlayer:0});
const put=object=>addObject(state,{owner:0,controller:0,...object},"battlefield");
const compiled=name=>{const result=compileScript(JSON.parse(readFileSync(new URL(name,import.meta.url))));assert.deepEqual(result.problems,[]);return result.definition;};
const mimic=compiled("../w0-w1-faces/game/engine/cards/m/metallic-mimic.json");
for(let index=0;index<2;index++){
 const old=addObject(state,{...mimic,card:"Metallic Mimic",owner:0,controller:0},"hand",0);
 beginRuleOperation(state,{kind:"move-one",id:old,to:"battlefield"});
 for(let step=0;state.awaiting&&step<10;step++){
  const choice=awaitingChoice(state),elf=choice.options.findIndex(option=>option.chosen==="Elf");
  answerRuleOperation(state,[elf>=0?elf:0]);
 }
 assert.equal(state.ruleOperation,undefined);
}
const holders=state.zones.battlefield.filter(id=>state.objects[id].card==="Metallic Mimic");
assert.deepEqual(holders.map(id=>state.objects[id].chosen),["Elf","Elf"],"both real loaded Mimics made the actual native entering type choice");
state=JSON.parse(JSON.stringify(state));
const sources=[0,1].map(index=>{const id=put({card:`Shared-ID Source ${index}`,types:["Battle"],defense:0});state.objects[id].protector=0;return id;});
const ids=["First Shared-ID Elf","Second Shared-ID Elf"].map(card=>addObject(state,{card,types:["Creature"],subtypes:["Elf"],power:2,toughness:2,owner:0,controller:0},"exile"));
put({...compiled("../w0-w1-faces/game/engine/cards/d/doubling-season.json"),card:"Doubling Season"});
put({card:"Second Counter Plus One",types:["Enchantment"],abilities:[{id:"second-plus",kind:"static",rule:"more-counters",counter:"+1/+1",plus:1,affects:{what:"permanent",controller:"you",named:"Second Shared-ID Elf"}}]});
state.exiledUntil=sources.map((source,index)=>({source,exiled:ids[index]}));afterResolving(state,[]);
try{
 for(let step=0;state.awaiting&&step<30;step++){
  const choice=awaitingChoice(state);state=JSON.parse(JSON.stringify(state));
  resolveAwaiting(state,choice.mode==="order"?choice.options.map((_,index)=>index):[0]);
 }
}catch(error){
 console.log(JSON.stringify({holders,progress:Object.values(state.ruleOperation?.sbaEntryProgress??{}).map(row=>({objectId:row.objectId,applied:row.frame.proposal.applied,appliedBy:row.frame.proposal.appliedBy}))}));throw error;
}
assert.equal(state.awaiting,null,"both actual returning Elves complete their native replacement choices");
const totals=["First Shared-ID Elf","Second Shared-ID Elf"].map(card=>state.objects[state.zones.battlefield.find(id=>state.objects[id].card===card)]?.counters["+1/+1"]);
assert.deepEqual(totals,[4,5],"each of the two distinct loaded Mimic sources applies once despite sharing an ability ID; the explicit second counter order remains authoritative");
