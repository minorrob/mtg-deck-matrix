    const setupCounts={};for(const e of setup)if(e.zone==='battlefield'&&e.seat===0)for(const name of e.cards)setupCounts[name]=(setupCounts[name]??0)+1;
    const changed=clone(original),used={},prefix=[];let reject=false;
    for(let i=0;i<(original.steps??[]).length;i++) {
      const step=original.steps[i];
      if(!Object.hasOwn(basic,step.tap)||(step.seat??0)!==0||Object.keys(step).some(k=>!['tap','seat','source','mana'].includes(k)))break;

      if(ordinal>=(setupCounts[step.tap]??0)||step.source&&JSON.stringify(step.source)!==JSON.stringify(source)){reject=true;break;}

      prefix.push({step:i,name:step.tap,ordinal,added:!Object.hasOwn(step,'source')});used[step.tap]=ordinal+1;
    }
    if(reject||!prefix.some(p=>p.added))continue;
    const roundtrip=clone(changed);for(const p of prefix)if(p.added)delete roundtrip.steps[p.step].source;same(roundtrip,original);
    candidates.push({file,card:data.card,fixtures,original,changed,prefix});
