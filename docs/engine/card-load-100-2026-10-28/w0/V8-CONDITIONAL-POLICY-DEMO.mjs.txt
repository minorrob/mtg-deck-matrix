function physical(state,actor,name,source){
 must(source&&typeof source==='object'&&Object.keys(source).every(k=>['zone','index','seat'].includes(k)),'malformed source');
 must(['battlefield','hand','library','exile','graveyard','command'].includes(source.zone)&&Number.isInteger(source.index)&&source.index>=0,'malformed source');
 must(source.seat===undefined||source.seat===actor,'wrong source actor');
 const raw=state.zones[source.zone],ids=['battlefield','exile'].includes(source.zone)?raw:raw[actor];
 const selected=ids.filter(id=>state.objects[id]?.card===name&&(source.zone==='battlefield'?controllerOf(state,id):state.objects[id].owner)===actor)[source.index];
 must(selected!==undefined,'missing physical source');return selected;
}
function chosenFrame(state,events,action,actor,name){
 const cast=events.filter(e=>e.kind==='GameEventSpellAbilityCast').map(fields).filter(f=>f.card?.cardId===action.objectId&&f.card.name===name&&f.si?.actor?.playerId===actor&&f.sa?.isSpell===(action.kind==='cast')&&(action.kind==='cast'||f.sa.abilityId===action.abilityId));
 must(cast.length===1,'missing or forged selected action event');
 const frame=state.stack.find(entry=>entry.stackId===cast[0].sa.stackId);
 must(frame&&frame.playerId===actor&&frame.name===name&&frame.kind===(action.kind==='cast'?'spell':'ability')&&(action.kind==='cast'||frame.abilityId===action.abilityId),'wrong selected native frame');
 must(state.stack.length===1&&state.stack.at(-1).stackId===frame.stackId,'intervening or extra frame');
 return frame;
}
function actIfLegal(state,step,afterApply=null){

 must(request&&Object.keys(request).every(k=>['cast','activate','tap','source','seat','ability','mana','targets'].includes(k)),'malformed action');

 must(typeof name==='string'&&name.trim()===name&&name.length>0&&Number.isInteger(actor)&&state.players[actor]&&!state.players[actor].lost,'malformed actor/card');
 must(['MAIN1','MAIN2'].includes(state.phase)&&state.priorityPlayer===actor&&!state.awaiting&&!state.ruleOperation&&!state.resolving&&!state.stack.length,'wrong stage or pending commitment');
 must(kind==='activate-mana'?(step.resolveSelected===undefined||step.resolveSelected===false):step.resolveSelected===true,'one expressly selected resolution required for nonmana action');
 must(kind==='activate-mana'||Array.isArray(request.targets)&&request.targets.length===0,'first proposal supports explicit targetless actions only');
 if(request.ability!==undefined)must(typeof request.ability==='string'&&request.ability.length>0,'malformed ability selector');
 if(request.mana!==undefined)must(request.mana&&Object.keys(request.mana).length>0&&Object.entries(request.mana).every(([c,n])=>'WUBRGC'.includes(c)&&c.length===1&&Number.isInteger(n)&&n>0),'malformed mana selector');
 const id=physical(state,actor,name,request.source),before=hashState(state);
 let actions=legalActions(state,actor).filter(a=>a.kind===kind&&a.objectId===id);
 if(request.ability!==undefined)actions=actions.filter(a=>a.abilityId===request.ability);
 if(request.mana!==undefined)actions=actions.filter(a=>hashState(a.mana)===hashState(request.mana));
 must(actions.length<=1,'ambiguous native action');
 if(!actions.length){must(hashState(state)===before,'query mutated state');return {branch:'unavailable',kind,actor,sourceId:id,events:[]};}
 const action=actions[0],events=applyAction(state,actor,action);
 if(afterApply)afterApply(state,events,action);
 must(!state.awaiting&&!state.ruleOperation&&!state.resolving,'action asked an unsupported mandatory question');
 if(kind==='activate-mana'){
   must(!state.stack.length,'mana action created an unsupported stack');
   must(events.some(e=>e.kind==='GameEventManaPool'&&fields(e).source?.cardId===id),'no selected native mana result');
   return {branch:'applied',kind,actor,sourceId:id,events};
 }
 const frame=chosenFrame(state,events,action,actor,name),stackId=frame.stackId;
 const resolved=resolveTop(state);
 must(!state.awaiting&&!state.ruleOperation&&!state.resolving&&!state.stack.some(e=>e.stackId===stackId),'selected resolution did not complete');
 must(resolved.some(e=>e.kind==='GameEventSpellResolved'&&fields(e).stackId===stackId&&fields(e).hasFizzled!==true),'no native successful resolution receipt');
 return {branch:'applied-and-resolved',kind,actor,sourceId:id,stackId,events:[...events,...resolved]};
}
function resolveAndCapture(state){

}
function answeredEntry(state,frame,events){
 must(!state.awaiting&&!state.ruleOperation&&!state.resolving&&!state.stack.some(e=>e.stackId===frame.stackId),'captured frame is not complete');

 must(moves.length===1,'no exact captured entrant movement');
}
function chooseEnteringIfAsked(state,token,request){
 must(contexts.has(token),'forged or missing resolution context');

 must(request&&Object.keys(request).every(k=>['card','seat','question','choose'].includes(k))&&request.card===frame.name&&request.seat===frame.playerId&&request.question==='entering-choose'&&Array.isArray(request.choose)&&request.choose.length===1&&typeof request.choose[0]==='string','wrong source/actor/question declaration');

 must(questionSignature!==null&&hashState(state.awaiting)===questionSignature,'stale changed question');

 must(state.awaiting.kind==='rule-operation'&&state.awaiting.player===frame.playerId&&q.id===`entering-choose:${frame.cardId}`&&q.mode==='one'&&q.min===1&&q.max===1&&q.options.every(o=>o.cardId===frame.cardId&&o.chosen===o.label)&&native?.kind==='entering'&&native.objectId===frame.cardId&&native.asks?.[0]?.choose==='creature type','another mandatory question is pending');

 return {branch:'answered',stackId:frame.stackId,sourceId:frame.cardId,choiceId:q.id,label:request.choose[0]};
}
