import assert from "node:assert/strict";

if(VARIANT==="entry"){
 assert.equal(state.zones.graveyard[0].some(id=>state.objects[id].card==="First Grant Walker"),true,"new native discard-instead entry condition redirects first return to graveyard when no land is available");
}else{
 const change=events.find(event=>event.kind==="GameEventCardChangeZone"&&event.data.fields.card.cardId===ids[0]);
 console.log(JSON.stringify({change,first:state.zones.battlefield.filter(id=>state.objects[id].card==="First Grant Walker")}));
 assert.equal(change?.data.fields.to.zoneType,"Exile","new live zone replacement must redirect the earlier prepared return");
 assert.notEqual(change.data.fields.becomes,ids[0],"CR400.8 re-exile creates a new exile identity");
