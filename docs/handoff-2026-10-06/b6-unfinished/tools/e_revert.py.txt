import re, sys
root = sys.argv[1]
def edit(rel, pairs):
    p = root + '/' + rel
    s = open(p).read()
    for old, new in pairs:
        assert old in s, (rel, old[:60])
        s = s.replace(old, new, 1)
    open(p, 'w').write(s)
