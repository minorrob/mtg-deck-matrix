def spell(text, effects, targets=None, **more):
    return {'kind': 'spell', 'text': text, 'targets': targets or [], 'effects': effects, **more}

def act(text, cost, effects, targets=None, **more):
    return {'kind': 'activated', 'text': text, 'cost': cost, **({'targets': targets} if targets else {}), 'effects': effects, **more}

def static(text, **more):
    return {'kind': 'static', 'text': text, **more}

MANA = lambda c: {'atom': 'mana', 'cost': c}
TAP = {'atom': '{T}'}
YOU_CREATURE = {'what': 'permanent', 'types': ['Creature'], 'controller': 'you'}
def tap(*lands, seat=None):
    return [{'tap': land, **({'seat': seat} if seat is not None else {})} for land in lands]
